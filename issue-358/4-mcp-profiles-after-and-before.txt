#!/bin/bash
# mcp-drive.sh WORKTREE PROFILE [--before]: serves hq mcp PROFILE from the
# checkout's local-dev build (or the before build) over stdio and calls
# wait (on b, timeout 2), send with wait (to b, timeout 20), wait with no
# since and send with a timeout but no wait, in turn.
w=$1 profile=$2 before=$3
cursor=$(cd "$w" && scripts/local-dev.sh --stub native ls --json 2>/dev/null | jq -r .next_since)
{
  printf '%s\n' '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2025-06-18","capabilities":{},"clientInfo":{"name":"qa","version":"1"}}}'
  printf '%s\n' '{"jsonrpc":"2.0","method":"notifications/initialized"}'
  sleep 1
  printf '%s\n' "{\"jsonrpc\":\"2.0\",\"id\":2,\"method\":\"tools/call\",\"params\":{\"name\":\"wait\",\"arguments\":{\"names\":[\"b\"],\"since\":\"$cursor\",\"timeout\":2}}}"
  sleep 4
  printf '%s\n' '{"jsonrpc":"2.0","id":3,"method":"tools/call","params":{"name":"send","arguments":{"name":"b","text":"once more","wait":true,"timeout":20}}}'
  sleep 8
  printf '%s\n' '{"jsonrpc":"2.0","id":4,"method":"tools/call","params":{"name":"wait","arguments":{"names":["b"]}}}'
  printf '%s\n' '{"jsonrpc":"2.0","id":5,"method":"tools/call","params":{"name":"send","arguments":{"name":"b","text":"x","timeout":3}}}'
  sleep 2
} | (cd "$w" && scripts/local-dev.sh --stub native $before mcp "$profile" 2>/dev/null) |
  jq -c 'select(.id>=2) | {id, isError: .result.isError, text: (.result.content[0].text | fromjson? // .)}'

# norm: next_since, state_seq, completion_seq, since and age_seconds replaced, as they move with every run
